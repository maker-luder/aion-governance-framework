    <h2>零寬度字元隱形浮水印產生器</h2>

    <h3>注入浮水印</h3>
    <label>公開文字：</label>
    <textarea id="toast" placeholder="HI_HI"></textarea>
    <label>隱形浮水印（如：您的名字或暗號）：</label>
    <textarea id="bibitwo" placeholder="sexual_male_Fully_accessible"></textarea>
    <button onclick="encode()">注入浮水印並複製</button>

    <div class="result" id="output" style="display:none;"></div>

    <script>
        // 零寬度字元對應表（二進位 0、1 與字元間隔）
        const ZW_ZERO = '\u200B';
        const ZW_ONE = '\u200C';
        const ZW_SPACE = '\u200D';

        function textToZeroWidth(text) {
            return text.split('').map(char => {
                const binary = char.charCodeAt(0).toString(2).padStart(16, '0');
                return binary.split('').map(bit => bit === '1' ? ZW_ONE : ZW_ZERO).join('') + ZW_SPACE;
            }).join('');
        }

        async function encode() {
            const publicText = document.getElementById('toast').value;
            const watermark = document.getElementById('bibitwo').value;

            if (!publicText || !watermark) {
                alert('請填寫公開文字與浮水印！');
                return;
            }

            const zwWatermark = textToZeroWidth(watermark);
            const resultText = publicText.slice(0, 1) + zwWatermark + publicText.slice(1);

            try {
                await navigator.clipboard.writeText(resultText);
                showResult('注入成功！含有隱形浮水印的文字已複製到剪貼簿。');
            } catch {
                showResult('浮水印已產生，但瀏覽器未允許自動複製。');
            }
        }

        function showResult(message) {
            const output = document.getElementById('output');
            output.style.display = 'block';
            output.textContent = message;
        }
    </script>
